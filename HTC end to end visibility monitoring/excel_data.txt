WITH base_boms AS
(
SELECT
eab.assmbl_db_id AS bom_db_id,
eab.assmbl_cd AS bom_assmbl_cd,
eab.assmbl_bom_id AS bom_id,

eab.assmbl_cd AS assmbl_cd,
eab.nh_assmbl_cd AS nh_assmbl_cd,
eab.bom_class_cd AS bom_class_cd,
eab.assmbl_bom_cd AS assmbl_bom_cd,
eab.assmbl_bom_name AS assmbl_bom_name

--,eab.creation_dt AS creation_dt,
--eab.revision_dt AS revision_dt

FROM
eqp_assmbl_bom eab

JOIN eqp_assmbl ea
ON ea.assmbl_cd = eab.nh_assmbl_cd
AND ea.assmbl_db_id = eab.nh_assmbl_db_id

WHERE
eab.bom_class_cd = 'TRK'
AND eab.bom_class_db_id = 0
AND ea.assmbl_class_cd IN ('ACFT', 'ENG')
AND eab.creation_dt > SYSDATE - 15
),
fg_events AS
(
SELECT DISTINCT
ei.assmbl_db_id AS fg_bom_db_id,
ei.assmbl_cd AS fg_bom_assmbl_cd,
ei.assmbl_bom_id AS fg_bom_id,

ee.event_id AS fg_event_id,
to_char(ee.event_dt , 'DD-MON-YYYY HH24:MI:SS') AS fg_event_dt,
--ee.event_dt AS fg_event_dt,
ee.event_gdt AS fg_event_gdt,
ee.event_sdesc AS fg_event_sdesc,
ee.event_status_cd AS fg_status_cd,

CASE
WHEN ee.event_status_cd = 'FGINST'
THEN 'INSTALL'
ELSE 'REMOVE'
END AS fg_event_type,

aircraft.inv_no_sdesc AS fg_aircraft,

inv.inv_no_db_id || ':' || inv.inv_no_id AS fg_inventory_key,
inv.inv_no_sdesc AS fg_inventory,
inv.barcode_sdesc AS fg_inv_barcode,
pos.eqp_pos_cd AS fg_config_slot,

pn.part_no_oem AS fg_part_no_oem,
pn.part_no_sdesc AS fg_part_no_sdesc,

ebp.bom_part_cd AS fg_part_group_cd,
ebp.bom_part_name AS fg_part_group_name,

rrr.remove_reason_cd AS fg_remove_reason,

hr.hr_cd AS fg_performed_by_hr_cd,

usr.username AS fg_performed_by_username,

CASE
WHEN usr.username IS NOT NULL
THEN usr.username || ' - ' || usr.first_name || ' ' || usr.last_name
WHEN hr.hr_cd IS NOT NULL
THEN hr.hr_cd
ELSE NULL
END AS fg_performed_by_user

FROM
evt_inv ei

JOIN base_boms b
ON b.bom_db_id = ei.assmbl_db_id
AND b.bom_assmbl_cd = ei.assmbl_cd
AND b.bom_id = ei.assmbl_bom_id

JOIN evt_event ee
ON ee.event_db_id = ei.event_db_id
AND ee.event_id = ei.event_id
AND ee.event_type_cd = 'FG'
AND ee.hist_bool = 1

JOIN inv_inv inv
ON inv.inv_no_db_id = ei.inv_no_db_id
AND inv.inv_no_id = ei.inv_no_id

JOIN inv_inv aircraft
ON aircraft.inv_no_db_id = ei.h_inv_no_db_id
AND aircraft.inv_no_id = ei.h_inv_no_id

LEFT JOIN eqp_part_no pn
ON pn.part_no_db_id = inv.part_no_db_id
AND pn.part_no_id = inv.part_no_id

LEFT JOIN eqp_part_baseline epb
ON epb.part_no_db_id = inv.part_no_db_id
AND epb.part_no_id = inv.part_no_id

LEFT JOIN eqp_bom_part ebp
ON ebp.bom_part_db_id = epb.bom_part_db_id
AND ebp.bom_part_id = epb.bom_part_id

LEFT JOIN eqp_assmbl_pos pos
ON pos.assmbl_db_id = ei.assmbl_db_id
AND pos.assmbl_cd = ei.assmbl_cd
AND pos.assmbl_bom_id = ei.assmbl_bom_id
AND pos.assmbl_pos_id = ei.assmbl_pos_id

LEFT JOIN sched_rmvd_part srp
ON srp.sched_db_id = ei.event_db_id
AND srp.sched_id = ei.event_id
AND srp.inv_no_db_id = ei.inv_no_db_id
AND srp.inv_no_id = ei.inv_no_id

LEFT JOIN ref_remove_reason rrr
ON rrr.remove_reason_db_id = srp.remove_reason_db_id
AND rrr.remove_reason_cd = srp.remove_reason_cd

LEFT JOIN org_hr hr
ON hr.hr_db_id = ee.editor_hr_db_id
AND hr.hr_id = ee.editor_hr_id

LEFT JOIN utl_user usr
ON usr.user_id = hr.user_id

WHERE
ei.main_inv_bool = 1
)
SELECT
b.assmbl_cd,
b.nh_assmbl_cd,
b.bom_class_cd,
b.assmbl_bom_cd,
b.assmbl_bom_name,
e.fg_part_group_cd,
e.fg_part_group_name,
--b.creation_dt,
--b.revision_dt,

e.fg_event_dt,
e.fg_event_type,
e.fg_status_cd,
e.fg_event_sdesc,

e.fg_aircraft,

e.fg_inventory_key,
e.fg_inventory,
e.fg_inv_barcode,
e.fg_config_slot,
e.fg_part_no_oem,
e.fg_part_no_sdesc,

e.fg_remove_reason,

e.fg_performed_by_user,
e.fg_performed_by_username,
e.fg_performed_by_hr_cd

FROM
base_boms b

LEFT JOIN fg_events e
ON e.fg_bom_db_id = b.bom_db_id
AND e.fg_bom_assmbl_cd = b.bom_assmbl_cd
AND e.fg_bom_id = b.bom_id

ORDER BY
--b.creation_dt DESC,
--b.assmbl_bom_cd,
e.fg_event_id DESC
